def fetch_snn_settings(max_retries=3, backoff_factor=2):
    """Fetches SNN settings from the Django API with retry logic."""
    retries = 0
    logging.info(f"Attempting to fetch SNN settings from {SNN_SETTINGS_API}")
    while retries < max_retries:
        try:
            response = requests.get(SNN_SETTINGS_API, timeout=10) # Added timeout
            response.raise_for_status() # Raise HTTPError for bad responses (4xx or 5xx)
            settings = response.json()
            logging.info(f"Successfully fetched SNN settings: {settings}")
            # Merge fetched settings with defaults, fetched values take precedence
            merged_settings = {**DEFAULT_SETTINGS, **settings}
            return merged_settings
        except requests.exceptions.Timeout:
            logging.warning(f"Timeout while fetching SNN settings (attempt {retries + 1}/{max_retries}).")
        except requests.exceptions.HTTPError as http_err:
            logging.error(f"HTTP error fetching SNN settings: {http_err} (attempt {retries + 1}/{max_retries}). Response: {response.text}")
        except requests.exceptions.RequestException as e:
            logging.error(f"Error fetching SNN settings: {e} (attempt {retries + 1}/{max_retries}).")

        retries += 1
        if retries < max_retries:
            wait_time = backoff_factor ** retries
            logging.info(f"Retrying to fetch SNN settings in {wait_time} seconds...")
            time.sleep(wait_time)

    logging.error(f"Max retries reached ({max_retries}). Using default SNN settings.")
    return DEFAULT_SETTINGS.copy()

    """Sends data to a specified API endpoint."""

            logging.info(f"Successfully sent {description}. Response: {response.status_code}")
            logging.debug(f"{description} Data: {json.dumps(data, indent=2)}")

    except requests.exceptions.RequestException as e:

        logging.debug(f"Data not sent: {json.dumps(data, indent=2)}")

def setup_snn_network(settings):
    """Initializes the Brian2 SNN model based on the provided settings."""
    # Network Parameters from settings
    N = settings["network_size"]
    F = settings["input_firing_rate"] * Hz
    vt_val = settings["firing_threshold"] * mV
    vr_val = settings["reset_potential"] * mV
    taupre_val = settings["tau_pre"] * ms
    taupost_val = settings["tau_post"] * ms
    gmax_val = settings["gmax"]
    synaptic_weight_base = settings["synaptic_weight"]
    tau_vt_val = settings["adaptive_threshold_tau"] * ms
    vth_increment_val = settings["adaptive_threshold_increment"] * mV
    refractory_period = settings["neuron_refractory_period"] * ms

    # Neuron Model: LIF with Adaptive Threshold
    # Note: Using explicit names in namespace to avoid Brian2 warnings if possible
    eqs = '''
    dv/dt = (El - v + R * ge * (Ee - v)) / taum : volt
    dge/dt = -ge / taue : 1
    dvt/dt = (vt_target - vt) / tau_vt_eq : volt (unless refractory) # Adaptive threshold dynamics
    '''

    # Neuron constants and parameters
    neuron_params = {
        'taum': 10 * ms, 'El': -65 * mV, 'Ee': 0 * mV, 'R': 10, 'taue': 5 * ms,
        'vt_target': vt_val, # Resting value for the threshold
        'tau_vt_eq': tau_vt_val, # Time constant for threshold adaptation
        'v_reset': vr_val, # Reset potential after spike
        'vth_spike_increment': vth_increment_val # Amount threshold increases after a spike
    }

    # Create NeuronGroup with explicit namespace
    # Threshold condition checks voltage against the adaptive threshold 'vt'
    # Reset statement resets voltage and increases the adaptive threshold 'vt'
    neurons = NeuronGroup(N, eqs,
                          threshold='v > vt',
                          reset='v = v_reset; vt += vth_spike_increment',
                          refractory=refractory_period,
                          method='euler',
                          namespace=neuron_params)
    neurons.v = neuron_params['El'] # Initialize voltage near resting potential
    neurons.vt = neuron_params['vt_target'] # Initialize adaptive threshold

    # STDP Parameters
    dApre_val = settings["learning_rate"] * gmax_val # Scale learning rate by gmax
    dApost_val = -dApre_val * taupre_val / taupost_val * 1.05 # Ensure depression slightly outweighs potentiation

    # Input Layer
    poisson_input = PoissonGroup(N, rates=F)

    # Synapses with STDP
    # w: synaptic weight
    # Apre/Apost: STDP trace variables
    synapse_model = '''
    w : 1 
    dApre/dt = -Apre / taupre_syn : 1 (event-driven)
    dApost/dt = -Apost / taupost_syn : 1 (event-driven)
    '''
    on_pre_actions = '''
    ge += w                 # Increase conductance of postsynaptic neuron
    Apre += dApre_syn       # Update presynaptic STDP variable
    w = clip(w + Apost, 0, gmax_syn) # Apply depression based on postsynaptic trace
    '''
    on_post_actions = '''
    Apost += dApost_syn     # Update postsynaptic STDP variable
    w = clip(w + Apre, 0, gmax_syn) # Apply potentiation based on presynaptic trace
    '''

    synapse_namespace = {
        'taupre_syn': taupre_val,
        'taupost_syn': taupost_val,
        'dApre_syn': dApre_val,
        'dApost_syn': dApost_val,
        'gmax_syn': gmax_val
    }

    S = Synapses(poisson_input, neurons,
                 model=synapse_model,
                 on_pre=on_pre_actions,
                 on_post=on_post_actions,
                 method='euler', # Can use 'linear' or 'euler'
                 namespace=synapse_namespace)

    # Connect all neurons except self-connections
    S.connect(condition='i != j')
    # Initialize weights: random component + base weight, clipped to [0, gmax]
    S.w = f'clip(rand() * {gmax_val} + {synaptic_weight_base}, 0, {gmax_val})'

    # Initialize stateful metrics
    neuron_energy_levels = np.ones(N) * settings["max_energy_level"]
    neuron_congestion_levels = np.zeros(N)

    # Monitors
    spike_monitor = SpikeMonitor(neurons)
    # Optional: Monitor state variables like voltage or weights
    # state_monitor = StateMonitor(neurons, ['v', 'vt'], record=True)
    # weight_monitor = StateMonitor(S, 'w', record=True) # Can be memory intensive

    return N, neurons, S, poisson_input, spike_monitor, neuron_energy_levels, neuron_congestion_levels #, state_monitor, weight_monitor


# --- Spike Train Analysis and Metrics ---

def analyze_spike_trains(spike_monitor, N, simulation_duration_sec):
    """Analyzes spike trains to extract firing rates, latencies, and ISIs."""
    spike_trains = spike_monitor.spike_trains() # Dictionary: neuron_index -> array of spike times
    if simulation_duration_sec <= 0:
        logging.warning("Simulation duration is zero or negative, cannot calculate rates.")
        simulation_duration_sec = 1.0 # Avoid division by zero

    firing_rates = np.zeros(N)
    # Use NaN for latency if no spike occurs, easier to handle than Inf in JSON/calculations
    latencies = np.full(N, np.nan)
    inter_spike_intervals = [[] for _ in range(N)] # List of lists for ISIs per neuron

    for neuron_idx in range(N):
        if neuron_idx in spike_trains:
            spikes = spike_trains[neuron_idx]
            num_spikes = len(spikes)

            if num_spikes > 0:
                # Firing rate (Hz)
                firing_rates[neuron_idx] = num_spikes / simulation_duration_sec

                # First spike latency (ms)
                latencies[neuron_idx] = spikes[0] / ms

                # Inter-spike intervals (ms)
                if num_spikes > 1:
                    intervals = np.diff(spikes) / ms
                    inter_spike_intervals[neuron_idx] = intervals.tolist() # Convert to list for JSON

    return {
        'firing_rates': firing_rates.tolist(), # Convert numpy arrays to lists for JSON
        'latencies': np.nan_to_num(latencies, nan=-1.0).tolist(), # Replace NaN with -1 for JSON
        'inter_spike_intervals': inter_spike_intervals
    }

def get_network_status(spike_monitor, N, neuron_energy_levels, neuron_congestion_levels, sim_time_seconds):
    """Calculates and returns the current network status summary."""
    spike_analysis = analyze_spike_trains(spike_monitor, N, sim_time_seconds)
    firing_rates = np.array(spike_analysis['firing_rates']) # Convert back to array for calculations
    latencies = np.array(spike_analysis['latencies']) # Has -1 where no spike occurred

    # FIX: Use python's sum() for generators, as warned
    total_packets = sum(len(st) for st in spike_monitor.spike_trains().values())
    active_nodes = sum(1 for fr in firing_rates if fr > 0) # FIX: Use python's sum()

    # Calculate average latency only over nodes that spiked
    valid_latencies = latencies[latencies != -1]
    avg_latency = np.mean(valid_latencies) if len(valid_latencies) > 0 else -1.0

    avg_energy = np.mean(neuron_energy_levels) if len(neuron_energy_levels) > 0 else 0.0
    avg_congestion = np.mean(neuron_congestion_levels) if len(neuron_congestion_levels) > 0 else 0.0

    return {
        "total_nodes": int(N),
        "active_nodes": int(active_nodes),
        "total_packets": int(total_packets),
        "avg_latency": float(avg_latency),  # RENAMED KEY
        "avg_energy": float(avg_energy),    # RENAMED KEY
        "avg_congestion": float(avg_congestion), # RENAME KEY (assuming you add avg_congestion to model)
        # Optional detailed data can remain if useful elsewhere, but won't go into NetworkStatus table directly
        "detailed_firing_rates_hz": spike_analysis['firing_rates'],
        "detailed_latencies_ms": spike_analysis['latencies'],
        "detailed_energy_levels": neuron_energy_levels.tolist(),
        "detailed_congestion_levels": neuron_congestion_levels.tolist(),
    }

def calculate_dynamic_routing_metrics(spike_analysis, neuron_energy_levels, neuron_congestion_levels, settings):
    """Calculates per-node scores used for routing decisions."""
    N = len(neuron_energy_levels)
    firing_rates = np.array(spike_analysis['firing_rates'])
    latencies = np.array(spike_analysis['latencies']) # Has -1 for non-spiking
    max_energy = settings["max_energy_level"]
    max_congestion = settings["max_congestion_level"]
    input_rate = settings["input_firing_rate"]

    detailed_scores_dict = {}
    for i in range(N):
        # Handle latency: Lower is better. Penalize non-spiking nodes (-1 latency).
        # Add epsilon to avoid division by zero if latency is somehow 0.
        latency_val = latencies[i]
        if latency_val == -1: # Node didn't spike, high cost
             latency_score_inv = 1e9 # Assign a very high inverse latency (bad score)
        else:
             latency_score_inv = latency_val + 1e-6 # Inverse latency (lower is better)


        # Handle energy: Higher remaining energy is better. Normalize.
        energy_score = (neuron_energy_levels[i] / max_energy) if max_energy > 0 else 0
        # Ensure score is within [0, 1]
        energy_score = np.clip(energy_score, 0, 1)

        # Handle congestion: Lower congestion is better. Normalize.
        congestion_val = neuron_congestion_levels[i]
        congestion_score_inv = congestion_val / max_congestion if max_congestion > 0 else 0
        # Ensure score is within [0, 1]
        congestion_score_inv = np.clip(congestion_score_inv, 0, 1)

        # Handle firing rate: May be good (activity) or bad (potential congestion).
        # Normalize by input rate. Let's assume moderate activity is okay.
        # A very high firing rate relative to input might indicate overload.
        firing_rate_score = (firing_rates[i] / input_rate) if input_rate > 0 else 0
        # Use a function that penalizes very high rates? e.g., Gaussian centered lower?
        # Simple linear for now: Higher rate = higher score component (potentially faster relay)
        firing_rate_score = np.clip(firing_rate_score, 0, 2) # Allow exceeding input rate slightly

        # Composite score: Weighted combination. Weights determine routing preference.
        # Weights should sum to 1 ideally, but relative values matter more.
        # Example: Prioritize low latency and high energy, penalize congestion.
        # Note: We use inverse latency and inverse congestion so lower values are better.
        # Score = w1*(1/latency) + w2*energy + w3*(1-congestion) + w4*firing_rate
        # Let's adjust the scoring logic to make higher scores always better.
        # latency_score: Higher score for lower latency. Normalize 1/latency? Maybe 1 / (1 + latency_ms/100)?
        # Let's use the previous formula's structure but clarify meaning.
        # Lower latency -> higher score contribution. Use max expected latency? Or just inverse.
        latency_contribution = 1.0 / (1 + latency_score_inv / 100.0) # Scale latency penalty, map -1 latency to near 0 score
        if latency_val == -1: latency_contribution = 0.0

        energy_contribution = energy_score # Higher energy -> higher score
        congestion_contribution = 1.0 - congestion_score_inv # Lower congestion -> higher score
        firing_rate_contribution = firing_rate_score # Simple linear contribution for now

        composite_score = (
            0.4 * latency_contribution +      # High weight for low latency
            0.3 * energy_contribution +       # Medium weight for high energy
            0.2 * congestion_contribution +   # Medium weight for low congestion
            0.1 * firing_rate_contribution    # Low weight for firing rate influence
        )

        detailed_scores_dict[i] = {
            "composite_score": composite_score,
            "latency_contribution": latency_contribution,
            "energy_contribution": energy_contribution,
            "congestion_contribution": congestion_contribution,
            "firing_rate_contribution": firing_rate_contribution,
            "raw_latency_ms": latencies[i],
            "raw_energy_level": neuron_energy_levels[i],
            "raw_congestion_level": neuron_congestion_levels[i],
            "raw_firing_rate_hz": firing_rates[i],
        }

    return detailed_scores_dict


def create_dynamic_graph(S, N, spike_analysis, neuron_congestion_levels, latencies_ms):
    """Creates a graph representation with dynamic edge weights."""
    dynamic_graph = {node: {} for node in range(N)} # Adjacency list: {source: {dest: weight}}

    # Get synaptic weights (consider them base cost)
    synaptic_weights = {}
    if len(S) > 0: # Check if synapses exist
      sources = S.i[:] # Get numpy array of source indices
      targets = S.j[:] # Get numpy array of target indices
      weights = S.w[:] # Get numpy array of weights
      for k in range(len(sources)):
          src, dest, w = int(sources[k]), int(targets[k]), float(weights[k])
          if src not in synaptic_weights: synaptic_weights[src] = {}
          synaptic_weights[src][dest] = w

    firing_rates = np.array(spike_analysis['firing_rates'])

    # Define edge weights (costs) dynamically
    for src in range(N):
        if src not in synaptic_weights: continue # Skip nodes with no outgoing synapses

        for dest, base_weight in synaptic_weights[src].items():
            # Cost increases with base weight (synaptic strength - stronger might mean more reliable?)
            # Cost increases with destination node's congestion
            # Cost increases with destination node's latency (if it spiked)
            # Cost might decrease with destination node's firing rate (activity)? Or increase (load)? Let's make it increase cost slightly.

            dest_congestion = neuron_congestion_levels[dest]
            dest_latency = latencies_ms[dest]
            dest_firing_rate = firing_rates[dest]

            # Combine factors: Higher values -> higher cost (weight)
            congestion_penalty = 1.0 + 0.5 * dest_congestion # Scale congestion impact
            latency_penalty = 1.0
            if dest_latency != -1: # Penalize based on latency only if node spiked
                latency_penalty = 1.0 + 0.1 * (dest_latency / 100.0) # Normalize latency impact (e.g., scale by 100ms)

            # Let's use the inverse of the base_weight (synaptic strength) as part of the cost
            # Stronger synapse (higher w) implies potentially faster/more reliable transmission -> lower cost component
            # Add epsilon to avoid division by zero if weight is 0
            synaptic_cost_factor = 1.0 / (base_weight + 1e-6)

            # Combine: Cost = Base(Synaptic) * Penalty(Congestion) * Penalty(Latency)
            # Base weight (S.w) is small (e.g., < 0.01). Let's rethink this.
            # Maybe cost = FixedBase + CongestionPenalty + LatencyPenalty - SynapticStrengthBonus?
            # Let's try cost = 1.0 + congestion + latency_penalty - weight*scale
            # Weight is small, need to scale it up to be meaningful vs congestion/latency.
            # Let's use a simpler multiplicative approach for now:
            dynamic_weight = (1.0 + base_weight) * congestion_penalty * latency_penalty

            # Ensure weight is positive
            dynamic_graph[src][dest] = np.maximum(0.01, dynamic_weight) # Use a small minimum cost

    return dynamic_graph


def dijkstra_dynamic_weights(start_node, end_node, dynamic_graph, N):
    """Finds the shortest path using Dijkstra on a graph with dynamic weights."""
    if start_node not in dynamic_graph or end_node not in dynamic_graph:
        logging.warning(f"Start ({start_node}) or End ({end_node}) node not in dynamic graph.")
        return None, float('inf') # No path possible

    distances = {node: float('inf') for node in range(N)}
    previous_nodes = {node: None for node in range(N)}
    distances[start_node] = 0
    priority_queue = [(0, start_node)] # (distance, node)

    while priority_queue:
        current_distance, current_vertex = heapq.heappop(priority_queue)

        # Node already processed with a shorter path
        if current_distance > distances[current_vertex]:
            continue

        # Explore neighbors
        if current_vertex in dynamic_graph:
            for neighbor, weight in dynamic_graph[current_vertex].items():
                distance = current_distance + weight
                # If found a shorter path to the neighbor
                if distance < distances[neighbor]:
                    distances[neighbor] = distance
                    previous_nodes[neighbor] = current_vertex
                    heapq.heappush(priority_queue, (distance, neighbor))

    # Reconstruct path
    path = []
    current_node = end_node
    if distances[current_node] == float('inf'):
        return None, float('inf') # No path found

    while current_node is not None:
        path.insert(0, current_node) # Insert at the beginning
        current_node = previous_nodes[current_node]

    if path[0] == start_node:
        return path, distances[end_node]
    else:
        return None, distances[end_node] # Path reconstruction failed (shouldn't happen if distance != inf)


def find_best_path_heuristic_guided(start_node, end_node, N, detailed_scores_dict, max_depth=10):
    """
    Finds a path using a heuristic best-first search based on composite node scores.
    This explores promising nodes first, guided by the SNN state.
    Uses recursion with depth limit to prevent infinite loops in case of cycles or suboptimal scores.
    Returns the path, its calculated score, and the history of scores considered at each step.
    """
    # Priority queue stores (-score, current_node, path_list, visited_set)
    # Use negative score because heapq is a min-heap, we want highest score first.
    # Start with the initial node score.
    start_score = detailed_scores_dict[start_node]['composite_score']
    queue = [(-start_score, start_node, [start_node])]
    # Keep track of visited states (node, path_prefix_tuple) to handle cycles and redundant exploration
    visited_paths = set()
    visited_paths.add( (start_node, tuple([start_node])) )
    # Store the best path found so far
    best_path_found = None
    best_path_score = -float('inf')

    # Store the scores evaluated for neighbors at each step for analysis/visualization
    neighbor_scores_history = {}

    while queue:
        neg_score, current_node, current_path = heapq.heappop(queue)
        current_score = -neg_score

        # Goal check
        if current_node == end_node:
             # Calculate path score (e.g., sum of node scores, or maybe average?)
             # Using sum might favor longer paths. Let's use average score.
             path_node_scores = [detailed_scores_dict[n]['composite_score'] for n in current_path]
             path_avg_score = sum(path_node_scores) / len(path_node_scores) if current_path else 0

             # If this path is better than the best found so far
             if path_avg_score > best_path_score:
                 best_path_score = path_avg_score
                 best_path_found = current_path
             # Continue searching in case there's an even better path via a different route
             # (A* would stop here if heuristic is admissible, but ours is just a heuristic)
             continue # Found a path to the end

        # Depth limit check
        if len(current_path) > max_depth:
            continue # Stop exploring this path if it gets too long

        # Explore neighbors (all other nodes for now, assuming full connectivity potential)
        # Sort neighbors by their composite score (descending)
        neighbor_options = []
        current_node_neighbor_scores = {}
        for neighbor_idx in range(N):
            if neighbor_idx != current_node: # Not the current node itself
                 neighbor_score = detailed_scores_dict[neighbor_idx]['composite_score']
                 neighbor_options.append((neighbor_score, neighbor_idx))
                 current_node_neighbor_scores[neighbor_idx] = detailed_scores_dict[neighbor_idx] # Store full details

        # Store the evaluated scores for this node's neighbors
        neighbor_scores_history[current_node] = current_node_neighbor_scores

        # Sort neighbors: highest score first
        neighbor_options.sort(key=lambda x: x[0], reverse=True)

        # Add valid neighbors to the queue
        for neighbor_score, neighbor_idx in neighbor_options:
            new_path = current_path + [neighbor_idx]
            path_tuple = tuple(new_path)
            state = (neighbor_idx, path_tuple)

            # Check if this state (node + path leading to it) was visited
            # Or simpler: check if neighbor is already in current_path to avoid simple cycles
            if neighbor_idx not in current_path: # and state not in visited_paths:
                # visited_paths.add(state) # Mark as visited if using state tracking
                # The score for the priority queue could be the score of the *neighbor*
                # or the accumulated score of the path so far. Let's use neighbor score for greedy best-first.
                heapq.heappush(queue, (-neighbor_score, neighbor_idx, new_path))

    # After exploring, return the best path found
    if best_path_found:
        # Recalculate final score based on the chosen path
        final_path_scores = [detailed_scores_dict[n]['composite_score'] for n in best_path_found]
        final_avg_score = sum(final_path_scores) / len(final_path_scores) if best_path_found else 0
        logging.info(f"Heuristic Search: Found path {best_path_found} with average score {final_avg_score:.3f}")
        return best_path_found, final_avg_score, neighbor_scores_history
    else:
        logging.warning("Heuristic Search: No path found to destination.")
        return None, -float('inf'), neighbor_scores_history

def run_simulation():
    """Runs the main SNN simulation and routing loop."""
    logging.info("--- Starting SNN Simulation ---")

    # Fetch initial settings
    snn_settings = fetch_snn_settings()
    logging.info(f"Using SNN settings: {json.dumps(snn_settings, indent=2)}")

    # Setup the SNN network
    N, neurons, S, poisson_input, spike_monitor, \
    neuron_energy_levels, neuron_congestion_levels = setup_snn_network(snn_settings)

    # Get simulation timing parameters
    # *** Ensure unit consistency based on fetch_snn_settings/API behavior ***
    simulation_step_duration = snn_settings["simulation_time"] * second # Assumes settings["simulation_time"] is in seconds
    routing_update_interval_sec = snn_settings["routing_update_interval"]

    # Get state update parameters from settings dictionary
    energy_consumption_per_spike = snn_settings["energy_consumption_per_spike"]
    energy_recovery_rate = snn_settings["energy_recovery_rate"]
    max_energy_level = snn_settings["max_energy_level"]
    congestion_increase_per_spike = snn_settings["congestion_increase_per_spike"]
    congestion_decay_rate = snn_settings["congestion_decay_rate"]
    max_congestion_level = snn_settings["max_congestion_level"]

    # Log the parameters being used for state updates
    logging.info(f"Using Energy Params: Consumption/Spike={energy_consumption_per_spike}, Recovery Rate={energy_recovery_rate}, Max={max_energy_level}")
    logging.info(f"Using Congestion Params: Increase/Spike={congestion_increase_per_spike}, Decay Rate={congestion_decay_rate}, Max={max_congestion_level}")


    loop_count = 0
    try:
        while True:
            loop_count += 1
            logging.info(f"\n--- Simulation Cycle {loop_count} ---")
            start_time = time.time()

            # Run Brian2 simulation for one step duration
            logging.info(f"Running Brian2 simulation for {simulation_step_duration}...")
            run(simulation_step_duration)
            brian_run_time = time.time() - start_time
            logging.info(f"Brian2 run finished in {brian_run_time:.2f} seconds.")

            # --- State Updates ---
            sim_duration_sec = float(simulation_step_duration / second)
            # Use count from the monitor which reflects spikes during the *last run*
            spike_counts_last_run = spike_monitor.count
            logging.debug(f"Spikes counted in last run: {spike_counts_last_run}")

            # Update Energy Levels
            energy_recovery = energy_recovery_rate * sim_duration_sec
            energy_consumption = spike_counts_last_run * energy_consumption_per_spike
            neuron_energy_levels += energy_recovery
            neuron_energy_levels -= energy_consumption
            neuron_energy_levels = np.clip(neuron_energy_levels, 0, max_energy_level)
            logging.debug(f"Updated Energy Levels: {np.round(neuron_energy_levels, 2)}")

            # Update Congestion Levels
            step_firing_rates = spike_counts_last_run / sim_duration_sec if sim_duration_sec > 0 else np.zeros(N)
            congestion_increase = step_firing_rates * sim_duration_sec * congestion_increase_per_spike
            # Simplified: congestion_increase = spike_counts_last_run * congestion_increase_per_spike
            decay_factor = np.power(1.0 - congestion_decay_rate, sim_duration_sec) # More precise decay over step
            neuron_congestion_levels *= decay_factor
            neuron_congestion_levels += congestion_increase
            neuron_congestion_levels = np.clip(neuron_congestion_levels, 0, max_congestion_level)
            logging.debug(f"Updated Congestion Levels: {np.round(neuron_congestion_levels, 3)}")

            # --- Analysis and Routing ---
            analysis_start_time = time.time()
            logging.info("Analyzing spike trains and calculating metrics...")
            # Analyze spikes over the last step duration for dynamic routing purposes
            spike_analysis = analyze_spike_trains(spike_monitor, N, sim_duration_sec)
            latencies_ms = np.array(spike_analysis['latencies']) # Used in dynamic graph

            # Calculate Node Scores (used by heuristic routing)
            detailed_scores_dict = calculate_dynamic_routing_metrics(spike_analysis, neuron_energy_levels, neuron_congestion_levels, snn_settings)

            # Calculate Network Status Summary (Using corrected function)
            network_status = get_network_status(spike_monitor, N, neuron_energy_levels, neuron_congestion_levels, sim_duration_sec)
            logging.info(f"Network Status Summary: TotalNodes={network_status['total_nodes']}, Active={network_status['active_nodes']}, Pkts={network_status['total_packets']}, AvgLatency={network_status['avg_latency']:.2f}, AvgEnergy={network_status['avg_energy']:.2f}, AvgCongestion={network_status['avg_congestion']:.2f}")


            # --- Routing Path Calculation ---
            source_node = 0
            dest_node = N - 1
            logging.info(f"Calculating routing paths from {source_node} to {dest_node}...")

            # 1. Dijkstra with Dynamic Weights
            logging.debug("Creating dynamic graph for Dijkstra...")
            dynamic_graph = create_dynamic_graph(S, N, spike_analysis, neuron_congestion_levels, latencies_ms)
            logging.debug(f"Dynamic Graph Weights Sample (Node 0): {dynamic_graph.get(0, {})}")

            logging.info("Running Dijkstra algorithm...")
            dijkstra_path, dijkstra_cost = dijkstra_dynamic_weights(source_node, dest_node, dynamic_graph, N)
            if dijkstra_path:
                logging.info(f"Dijkstra Path Found: {dijkstra_path} (Cost: {dijkstra_cost:.4f})")
            else:
                logging.warning("Dijkstra Algorithm: No path found.")

            # 2. Heuristic Best-First Search (using node scores)
            logging.info("Running Heuristic Best-First Search...")
            heuristic_path, heuristic_score, neighbor_scores = find_best_path_heuristic_guided(
                source_node, dest_node, N, detailed_scores_dict, max_depth=N*2
            )
            if heuristic_path:
                 logging.info(f"Heuristic Path Found: {heuristic_path} (Avg Score: {heuristic_score:.3f})")
            # neighbor_scores contains the history useful for explanation

            analysis_routing_time = time.time() - analysis_start_time
            logging.info(f"Analysis and routing calculations finished in {analysis_routing_time:.2f} seconds.")


            # --- API Communication ---
            api_start_time = time.time()

            # --- Prepare Payload for /api/simulation_data/ endpoint ---
            # This payload structure matches the receive_simulation_data view
            logging.info("Preparing simulation data payload...")

            routing_path_data_to_send = {
                "selected_path": heuristic_path if heuristic_path else [],
                "stdp_path_score": heuristic_score if heuristic_path else None,
                "dijkstra_path": dijkstra_path if dijkstra_path else [],
                "neighbor_scores_history": neighbor_scores if neighbor_scores else {},
                "source_node": source_node,
                "destination_node": dest_node,
            }

            status_data_to_send = {
                "timestamp": datetime.datetime.now().isoformat(), # Add timestamp for status saving consistency
                "total_nodes": network_status['total_nodes'],
                "active_nodes": network_status['active_nodes'],
                "total_packets": network_status['total_packets'],
                "avg_latency": network_status['avg_latency'],
                "avg_energy": network_status['avg_energy'],
                "avg_congestion": network_status['avg_congestion'] # Assumes field exists in model
            }

            simulation_data_payload = {
                "timestamp": datetime.datetime.now().isoformat(),
                "simulation_time_ms": sim_duration_sec * 1000.0, # Correct key and unit
                # Send lists directly, view handles JSON conversion for saving
                "firing_rates": network_status['detailed_firing_rates_hz'],
                "latencies_ms": network_status['detailed_latencies_ms'],
                "energy_levels": network_status['detailed_energy_levels'],
                "congestion_levels": network_status['detailed_congestion_levels'],
                # Nested dictionaries expected by the view
                "routing_path_data": routing_path_data_to_send,
                "status_data": status_data_to_send,
                # Optional extra info not directly saved by receive_simulation_data view
                "cycle_number": loop_count,
                "brian_run_time_s": brian_run_time,
                "analysis_routing_time_s": analysis_routing_time,
            }

            logging.info("Sending detailed simulation data to API (includes routing and status)...")
            logging.debug(f"Payload N={N}, firing_rates length={len(simulation_data_payload['firing_rates'])}")
            logging.debug(f"Payload latencies_ms length={len(simulation_data_payload['latencies_ms'])}")
            logging.debug(f"Payload energy_levels length={len(simulation_data_payload['energy_levels'])}")
            logging.debug(f"Payload congestion_levels length={len(simulation_data_payload['congestion_levels'])}")
            send_api_data(SEND_SIMULATION_DATA_API, simulation_data_payload, "simulation data")

            api_time = time.time() - api_start_time
            logging.info(f"API communication finished in {api_time:.2f} seconds.")

            logging.info(f"Waiting for {routing_update_interval_sec} seconds...")
            time.sleep(routing_update_interval_sec)

    except KeyboardInterrupt:
        logging.info("Simulation stopped by user (KeyboardInterrupt).")
    except Exception as e:
        logging.error(f"An unexpected error occurred in run_simulation: {e}", exc_info=True) # Log traceback
    finally:
        logging.info("--- Simulation Ended ---")
