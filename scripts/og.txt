# ✅ Handle API Failures Gracefully: Retry with exponential backoff
def fetch_snn_settings(max_retries=3, backoff_factor=2):
    retries = 0
    while retries < max_retries:
        try:
            response = requests.get(SNN_SETTINGS_API)
            if response.status_code == 200:
                return response.json()
            else:
                logging.error(f"Failed to fetch SNN settings, status code: {response.status_code}")
            #pass
        except requests.exceptions.RequestException as e:
            logging.error(f"Error fetching SNN settings: {e}")

        retries += 1
        wait_time = backoff_factor ** retries
        logging.info(f"Retrying to fetch SNN settings in {wait_time} seconds...")
        time.sleep(wait_time)

    logging.error("Max retries reached, using default SNN settings.")
    return {} # Return empty dict on failure

# Find best path using Dijkstra with dynamic edge weights
def dijkstra_dynamic_weights(spike_monitor, graph, start, end, S): # Make sure to accept spike_monitor
    queue = [(0, start)]
    distances = {i: float('inf') for i in graph}
    distances[start] = 0
    previous_nodes = {i: None for i in graph}

    while queue:
        current_distance, current_vertex = heapq.heappop(queue)

        if current_distance > distances[current_vertex]:
            continue

        # ✅ Dijkstra Algorithm Limitation: Recalculate edge weights dynamically
        dynamic_graph = create_dynamic_graph(spike_monitor, S, graph, neuron_congestion_levels) # Pass congestion levels and spike_monitor
        if current_vertex not in dynamic_graph: # Handle cases where a node might have no outgoing edges in dynamic graph
            continue

        for neighbor, weight in dynamic_graph[current_vertex].items():
            distance = current_distance + weight
            if distance < distances[neighbor]:
                distances[neighbor] = distance
                previous_nodes[neighbor] = current_vertex
                heapq.heappush(queue, (distance, neighbor))

    path = []
    current_node = end
    while current_node is not None:
        path.append(current_node)
        current_node = previous_nodes[current_node]
    path.reverse()
    return path

def create_dynamic_graph(spike_monitor, S, original_graph, neuron_congestion_levels): # Make sure to accept spike_monitor
    dynamic_graph = {node: {} for node in original_graph}
    spike_trains = spike_monitor.spike_trains()
    firing_rates = [len(spike_trains[i]) / (simulation_time / second) if i in spike_trains else 0 for i in range(N)]
    latencies = [spike_trains[i][0] / ms if i in spike_trains and len(spike_trains[i]) > 0 else np.inf for i in range(N)]
    congestion = neuron_congestion_levels # Use stateful congestion

    for i in range(len(S)):
        src, dest = int(S.i[i]), int(S.j[i])
        weight = float(S.w[i])

        # ✅ Dijkstra Algorithm Limitation: Edge weights based on synaptic strength, latency, congestion, firing rate
        dynamic_weight = weight * (1 + 0.1 * firing_rates[dest] - 0.05 * congestion[dest] - 0.05 * (latencies[dest] if latencies[dest] != np.inf else 0)) # Example dynamic weight formula
        dynamic_graph[src][dest] = np.maximum(0.0, dynamic_weight) # Ensure weight is not negative

    return dynamic_graph

def find_best_path_stdp_guided(spike_monitor, source, destination, S, graph, neuron_energy_levels, neuron_congestion_levels, visited=None, current_path=None, neighbor_scores_history=None): # Make sure to accept spike_monitor
    # Removed SpikeMonitor instantiation from here - using the one passed as argument
    if visited is None:
        visited = set()
    if current_path is None:
        current_path = [source]
    if neighbor_scores_history is None: # Initialize if not passed
        neighbor_scores_history = {}

    if source == destination:

        path_score = calculate_path_score(spike_monitor, current_path, S, neuron_energy_levels, neuron_congestion_levels)
        logging.debug(f"  Reached destination {destination} via path: {current_path}, Score: {path_score:.2f}")
        return current_path, path_score, neighbor_scores_history # Return history

    visited.add(source)
    best_path = None
    best_score = -np.inf  # Initialize with negative infinity
    run(100 * ms)
    # 1. Create Dictionary & Get Neighbor Parameters
    detailed_scores_dict = calculate_dynamic_routing_metrics(spike_monitor, source, destination, S, neuron_energy_levels, neuron_congestion_levels) # Pass spike_monitor
    #logging.debug(f"  Router {source}: Created neighbor parameter dictionary: {json.dumps(detailed_scores_dict, indent=2)}") # Log the dictionary

    neighbor_scores = [(i, detailed_scores_dict[i]["composite_score"]) for i in range(N) if i != source and i not in visited] # Use dict for neighbor scores
    sorted_neighbors = sorted(neighbor_scores, key=lambda x: x[1], reverse=True) # Prioritize higher scores

    # Store neighbor scores for current node in history
    neighbor_scores_history[source] = detailed_scores_dict  # Store the dict

    logging.debug(f"Exploring paths from node {source}. Considered neighbors: {[(n, score) for n, score in sorted_neighbors]}")

    for neighbor_index, neighbor_score in sorted_neighbors:
        neighbor = neighbor_index
        if neighbor not in visited: # Ensure not revisiting
            logging.debug(f"  Router {source}: Exploring neighbor {neighbor} (score: {neighbor_score:.2f})")
            # 4. Choose the Best Composite (Implicitly by sorted_neighbors and loop order)
            logging.debug(f"  Router {source}: Choosing neighbor {neighbor} with best composite score: {neighbor_score:.2f}") # Highlight choice

            # 5. Spike and Check New Neighbors (Recursively)
            next_path, path_score, updated_history = find_best_path_stdp_guided(spike_monitor, neighbor, destination, S, graph, neuron_energy_levels, neuron_congestion_levels, visited, current_path + [neighbor], neighbor_scores_history) # Pass spike_monitor and history
            neighbor_scores_history = updated_history # Update history from recursive call
            if next_path: # Found a path from this neighbor
                logging.debug(f"  Path found through neighbor {neighbor} from {source} to {destination}: {next_path}, Score: {path_score:.2f}")
                if path_score > best_score:
                    logging.debug(f"    Path through {neighbor} is better (score: {path_score:.2f} > current best: {best_score:.2f}). Updating best path.")
                    best_score = path_score
                    best_path = next_path
                else:
                    logging.debug(f"    Path through {neighbor} is NOT better (score: {path_score:.2f} <= current best: {best_score:.2f}). Rejecting path.") # Rejection reason: Score not better
            else:
                logging.debug(f"  No path found through neighbor {neighbor} from {source}. Rejecting neighbor path.") # Rejection reason: No path downstream
        else:
            logging.debug(f"  Neighbor {neighbor} already visited from {source}. Rejecting neighbor (already visited).") # Rejection reason: Already visited

    if best_path:
        logging.debug(f"Router {source}: Selected best path: {best_path} with score: {best_score:.2f}") # Log selected best path from current node
    else:
        logging.debug(f"Router {source}: No path found to destination.") # Log if no path found from current node

    visited.remove(source) # Backtrack: Allow source to be visited in other paths
    return best_path, best_score, neighbor_scores_history # Return history


def calculate_path_score(spike_monitor, path, S, neuron_energy_levels, neuron_congestion_levels): # Make sure to accept spike_monitor
    total_score = 0
    for node in path:
        detailed_scores = calculate_dynamic_routing_metrics(spike_monitor, node, path[-1], S, neuron_energy_levels, neuron_congestion_levels) # Pass spike_monitor
        # ✅ Extract composite_score from detailed_scores for the current node
        node_score = detailed_scores[node]["composite_score"]  # Access composite_score for 'node' itself

        total_score += node_score # Sum the composite score of each node in the path
    return total_score
